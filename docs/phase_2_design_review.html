Output Contract (`OrchestratorDecision` with `reasonCode`)

export type InterviewAction =
  | "OPEN_INTERVIEW"         // Turn 0: Persona intro & background request
  | "ACKNOWLEDGE_INTRO"      // Turn 1: Anchor to candidate's intro & ask Pillar 1
  | "HANDLE_GREETING"        // Greeting: brief warm phrase + repeat question
  | "PROMPT_FOR_SUBSTANCE"   // Non-substantive 1: Reprompt patiently without validation
  | "PIVOT_AWAY"             // Non-substantive 2+: Release topic & switch pillar
  | "ANSWER_CLARIFICATION"   // Candidate asked question: directive for LLM to answer in 1-2 sentences
  | "HANDLE_META_REQUEST"    // Candidate meta-instruction: put candidate in hot seat
  | "DRILL_DOWN"             // Substantive 1: probe tradeoff, mechanics, concurrency
  | "CHALLENGE_TRADEOFF"     // Substantive 2: challenge failure modes & constraints
  | "ADVANCE_PILLAR"         // Substantive 3+: transition to next domain pillar
  | "CONCLUDE_ROUND";        // Wrap up requested or round complete: emit [SESSION_COMPLETED]

export type DecisionReasonCode =
  | "INITIAL_OPENING"
  | "CANDIDATE_REQUESTED_END"
  | "GREETING_DETECTED"
  | "CANDIDATE_META_INSTRUCTION"
  | "CANDIDATE_ASKED_CLARIFICATION"
  | "NON_SUBSTANTIVE_REPROMPT"
  | "NON_SUBSTANTIVE_PIVOT_LIMIT"
  | "CANDIDATE_INTRO_RECEIVED"
  | "SUBSTANTIVE_PROBE_DEPTH"
  | "SUBSTANTIVE_CHALLENGE_TRADEOFF"
  | "PILLAR_PROGRESSION_ADVANCE"
  | "ALL_PILLARS_EXHAUSTED";

export interface OrchestratorDecision {
  action: InterviewAction;
  reasonCode: DecisionReasonCode;
  targetPillarIndex: number;
  targetPillarName: string;
  consecutiveTurnsOnPillar: number;
  isTerminalTurn: boolean;
  directiveInstruction: string;
  groundingKeywords: string[];
}

4. Clarification Boundary & Pluggable Progression Architecture

Strict Clarification Boundary (`ANSWER_CLARIFICATION`)

To preserve clean architectural boundaries:

  • The Orchestrator identifies that the candidate asked a question and outputs a directive: "The candidate asked a clarification regarding the architecture. Answer knowledgeably in 1–2 sentences, then resume the interview scenario."
  • The Orchestrator NEVER hard-codes technical answers.
  • The LLM synthesizes the natural technical explanation within the persona's voice and background.

Pluggable Progression Architecture (Phase 2 → Phase 3 Evolution)

Phase 2 Baseline → Phase 3 Evaluation Hooks
In Phase 2, the progression policy utilizes the turn-count heuristic (2 substantive turns per pillar). The method shouldAdvancePillar(state, options) is modularized so that in Phase 3, real-time rubric scores (e.g. candidate demonstrated mastery after 1 turn or requires remediation) can dynamically adjust the threshold without refactoring the orchestrator core.
